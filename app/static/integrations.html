# Optional PreToolUse hook that POSTs traces to hosted /v1/trace
# (replace HOST with your PUBLIC_BASE_URL)

  -H "Authorization: Bearer $OPENAI_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"title":"hook","tool_events":[{"tool":"Read","args":{"path":"x"}}]}'
